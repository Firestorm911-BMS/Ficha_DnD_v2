      const { saveSpellModal } = await import('../src/modules/spell-modal.js');
      section('spell-modal — ritual + saveSpellModal (FEAT-08)');

      function _fillSpellForm(overrides = {}) {
        const f = {
          smName:        'Detectar Magia',
          smLevel:       '1',
          smSchool:      'Adivinación',
          smCastTime:    '1 acción',
          smRange:       'Personal',
          smComponents:  'V, S',
          smDuration:    'Conc., 10 min',
          smConc:        false,
          smRitual:      false,
          smSave:        '',
          smAttack:      '',
          smCastAttr:    '',
          smDesc:        'Detectas magia a 9m.',
          ...overrides,
        };
        for (const [id, val] of Object.entries(f)) {
          const el = document.getElementById(id);
          if (!el) continue;
          if (el.type === 'checkbox') el.checked = val;
          else el.value = val;
        }
      }

      test('saveSpellModal guarda ritual:false por defecto', () => {
        state.spells.length = 0;
        _fillSpellForm({ smRitual: false });
        saveSpellModal();
        const sp = state.spells[0];
        assert(sp !== undefined,      'debe existir el conjuro guardado');
        assertEqual(sp.ritual, false, 'ritual debe ser false');
      });

      test('saveSpellModal guarda sourceClass cuando se elige clase', () => {
        state.spells.length = 0;
        document.getElementById('smSourceClass').value = 'Mago';
        _fillSpellForm({ smName: 'Misil Mágico', smLevel: '1' });
        saveSpellModal();
        const sp = state.spells[0];
        assert(sp !== undefined,               'debe existir el conjuro');
        assertEqual(sp.sourceClass, 'Mago',    'sourceClass debe ser Mago');
        document.getElementById('smSourceClass').value = '';
      });

      test('saveSpellModal guarda ritual:true cuando el checkbox está marcado', () => {
        state.spells.length = 0;
        _fillSpellForm({ smRitual: true });
        saveSpellModal();
        const sp = state.spells[0];
        assert(sp !== undefined,     'debe existir el conjuro guardado');
        assertEqual(sp.ritual, true, 'ritual debe ser true');
      });

      test('saveSpellModal incluye concentration cuando smConc está marcado', () => {
        state.spells.length = 0;
        _fillSpellForm({ smConc: true, smRitual: true });
        saveSpellModal();
        const sp = state.spells[0];
        assert(sp !== undefined,           'debe existir el conjuro guardado');
        assertEqual(sp.concentration, true, 'concentration debe ser true');
        assertEqual(sp.ritual, true,        'ritual debe ser true');
      });
